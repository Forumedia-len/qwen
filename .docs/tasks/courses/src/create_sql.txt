-- ────────────────────────────────────────────────────────────────
-- 1. at_base_active_court_course_groups
-- Основная сущность: курс (логическая группа бронирований)
-- ────────────────────────────────────────────────────────────────
CREATE TABLE `at_base_active_court_course_groups` (
  `course_group_id` int(11) NOT NULL AUTO_INCREMENT COMMENT 'Уникальный ID курса',
  `title` varchar(255) NOT NULL COMMENT 'Название курса (обязательно)',
  `description` mediumtext DEFAULT NULL COMMENT 'Описание курса (HTML / plain)',
  `image_path` varchar(512) DEFAULT NULL COMMENT 'Путь к изображению (одно). Формат: /uploads/kurse/xxx.jpg',
  `trainer_client_id` int(10) UNSIGNED NOT NULL COMMENT 'Тренер — ссылка на clients.client_id (обязательно клиент)',
  `area_id` int(10) UNSIGNED DEFAULT NULL COMMENT 'Фактическое место проведения (areas.area_id). NULL = "Anderer Ort"',
  `is_other_location` enum('0','1') NOT NULL DEFAULT '0' COMMENT '1 — курс НЕ отображается в календаре площадок',
  `min_participants` int(4) UNSIGNED NOT NULL DEFAULT 1 COMMENT 'Минимум участников для запуска',
  `max_participants` int(4) UNSIGNED NOT NULL DEFAULT 10 COMMENT 'Жёсткий лимит участников',
  `is_sessions_bookable_separately` enum('0','1') NOT NULL DEFAULT '0' COMMENT '1 — можно бронировать отдельные занятия',
  `payment_type` enum('self','trainer','free') NOT NULL DEFAULT 'self' COMMENT 'self: каждый платит; trainer: платит тренер; free: бесплатно',
  `trainer_price` double(10,2) DEFAULT NULL COMMENT 'Цена для тренера (только если payment_type = "trainer")',
  `cancel_days_limit` int(3) UNSIGNED NOT NULL DEFAULT 3 COMMENT 'Сколько дней до старта можно отменить (участником)',
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Дата создания курса',
  `created_by_user_id` int(11) DEFAULT NULL COMMENT 'Кто создал: users.user_id (NULL = системное создание)',
  `status` enum('active','cancelled','archived') NOT NULL DEFAULT 'active' COMMENT 'active — виден; cancelled — уведомление отправлено; archived — в архиве',
  PRIMARY KEY (`course_group_id`),
  KEY `idx_trainer_client_id` (`trainer_client_id`),
  KEY `idx_area_id` (`area_id`),
  KEY `idx_created_by_user_id` (`created_by_user_id`),
  CONSTRAINT `fk_cg_trainer_client` FOREIGN KEY (`trainer_client_id`) REFERENCES `at_base_active_court_clients` (`client_id`) ON DELETE RESTRICT,
  CONSTRAINT `fk_cg_area` FOREIGN KEY (`area_id`) REFERENCES `at_base_active_court_areas` (`area_id`) ON DELETE SET NULL,
  CONSTRAINT `fk_cg_created_by_user` FOREIGN KEY (`created_by_user_id`) REFERENCES `at_base_active_court_users` (`user_id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Группы курсов (Kurse) — основная сущность';

-- ────────────────────────────────────────────────────────────────
-- 2. at_base_active_court_course_prices
-- Цены: НОРМАЛИЗОВАННАЯ версия — поддержка любого числа membership_type
-- ────────────────────────────────────────────────────────────────
CREATE TABLE `at_base_active_court_course_prices` (
  `course_group_id` int(11) NOT NULL COMMENT 'Курс',
  `membership_type_id` int(11) NOT NULL COMMENT 'Тип членства (clients.membership_type_id)',
  `price_full` double(10,2) DEFAULT NULL COMMENT 'Цена за весь курс. NULL = недоступно. 0.00 = бесплатно.',
  `price_single` double(10,2) DEFAULT NULL COMMENT 'Цена за одно занятие (только если is_sessions_bookable_separately = 1)',
  PRIMARY KEY (`course_group_id`, `membership_type_id`),
  KEY `idx_membership_type_id` (`membership_type_id`),
  CONSTRAINT `fk_cp_course_group` FOREIGN KEY (`course_group_id`) REFERENCES `at_base_active_court_course_groups` (`course_group_id`) ON DELETE CASCADE,
  CONSTRAINT `fk_cp_membership_type` FOREIGN KEY (`membership_type_id`) REFERENCES `at_base_active_court_membership_types` (`membership_type_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Цены на курсы — по типам членства (гибко, масштабируемо)';

-- ────────────────────────────────────────────────────────────────
-- 3. at_base_active_court_course_reservations
-- Связь: reservation → курс. Одна строка = одно занятие.
-- Важно: reservation.area_id = 99999, если is_other_location = 1
-- ────────────────────────────────────────────────────────────────
CREATE TABLE `at_base_active_court_course_reservations` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT 'Вспомогательный PK',
  `course_group_id` int(11) NOT NULL COMMENT 'Курс',
  `reservation_id` int(11) UNSIGNED NOT NULL COMMENT 'Бронь (reservations.reservation_id)',
  `is_full_course_reservation` enum('0','1') NOT NULL DEFAULT '1' COMMENT '1 — забронирован весь курс; 0 — частичное участие (ручное)',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_reservation_id` (`reservation_id`),  -- каждая бронь — только в одном курсе
  KEY `idx_course_group_id` (`course_group_id`),
  CONSTRAINT `fk_cr_course_group` FOREIGN KEY (`course_group_id`) REFERENCES `at_base_active_court_course_groups` (`course_group_id`) ON DELETE CASCADE,
  CONSTRAINT `fk_cr_reservation` FOREIGN KEY (`reservation_id`) REFERENCES `at_base_active_court_reservations` (`reservation_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Связь курсов с бронированиями (одна запись = одно занятие)';

-- ────────────────────────────────────────────────────────────────
-- 4. at_base_active_court_course_participant_overrides
-- 🆕 РАСШИРЕННАЯ ВЕРСИЯ: поддержка цен на (курс, участник, занятие)
-- Удалён UNIQUE (course_group_id, client_id) → возможны N записей
-- ────────────────────────────────────────────────────────────────
CREATE TABLE `at_base_active_court_course_participant_overrides` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT 'PK',
  `course_group_id` int(11) NOT NULL COMMENT 'Курс',
  `client_id` int(10) UNSIGNED NOT NULL COMMENT 'Участник',
  `reservation_id` int(11) UNSIGNED DEFAULT NULL COMMENT 'Если NULL — переопределение для всего курса; иначе — для одного занятия',
  `price_full_override` double(10,2) DEFAULT NULL COMMENT 'Цена за весь курс (только если reservation_id IS NULL)',
  `price_single_override` double(10,2) DEFAULT NULL COMMENT 'Цена за одно занятие (только если reservation_id IS NOT NULL)',
  `created_by_user_id` int(11) DEFAULT NULL COMMENT 'Кто установил: user_id (админ или тренер)',
  `created_at` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Когда создана запись',
  PRIMARY KEY (`id`),
  -- ❗ НЕТ UNIQUE(course_group_id, client_id) — теперь несколько записей на одного участника допустимы
  KEY `idx_course_client` (`course_group_id`, `client_id`),
  KEY `idx_reservation_id` (`reservation_id`),
  KEY `idx_client_id` (`client_id`),
  CONSTRAINT `fk_cpo_course_group` FOREIGN KEY (`course_group_id`) REFERENCES `at_base_active_court_course_groups` (`course_group_id`) ON DELETE CASCADE,
  CONSTRAINT `fk_cpo_client` FOREIGN KEY (`client_id`) REFERENCES `at_base_active_court_clients` (`client_id`) ON DELETE CASCADE,
  CONSTRAINT `fk_cpo_reservation` FOREIGN KEY (`reservation_id`) REFERENCES `at_base_active_court_reservations` (`reservation_id`) ON DELETE CASCADE,
  CONSTRAINT `fk_cpo_user` FOREIGN KEY (`created_by_user_id`) REFERENCES `at_base_active_court_users` (`user_id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Персональные цены: (курс, участник) ИЛИ (курс, участник, занятие)';

-- ────────────────────────────────────────────────────────────────
-- 5. at_base_active_court_course_participant_allowed_sessions
-- Дополнительные ограничения при частичном участии
-- ────────────────────────────────────────────────────────────────
CREATE TABLE `at_base_active_court_course_participant_allowed_sessions` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT 'PK',
  `override_id` int(11) NOT NULL COMMENT 'Ссылка на course_participant_overrides.id',
  `reservation_id` int(11) UNSIGNED NOT NULL COMMENT 'Разрешённое занятие',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_override_reservation` (`override_id`, `reservation_id`),
  KEY `idx_reservation_id` (`reservation_id`),
  CONSTRAINT `fk_cpas_override` FOREIGN KEY (`override_id`) REFERENCES `at_base_active_court_course_participant_overrides` (`id`) ON DELETE CASCADE,
  CONSTRAINT `fk_cpas_reservation` FOREIGN KEY (`reservation_id`) REFERENCES `at_base_active_court_reservations` (`reservation_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Разрешённые занятия для участника (только при частичном режиме)';

-- ────────────────────────────────────────────────────────────────
-- 6. at_base_active_court_user_course_roles
-- Роль "Kurs-Admin" (Level 4) — без изменения users
-- ────────────────────────────────────────────────────────────────
CREATE TABLE `at_base_active_court_user_course_roles` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT 'PK',
  `user_id` int(11) NOT NULL COMMENT 'Пользователь (обычно тренер)',
  `course_group_id` int(11) NOT NULL COMMENT 'Курс',
  `role` enum('admin') NOT NULL DEFAULT 'admin' COMMENT 'Только "admin" на данный момент',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_user_course` (`user_id`, `course_group_id`),
  KEY `idx_course_group_id` (`course_group_id`),
  CONSTRAINT `fk_ucr_user` FOREIGN KEY (`user_id`) REFERENCES `at_base_active_court_users` (`user_id`) ON DELETE CASCADE,
  CONSTRAINT `fk_ucr_course` FOREIGN KEY (`course_group_id`) REFERENCES `at_base_active_court_course_groups` (`course_group_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Права тренеров (Kurs-Admin)';
